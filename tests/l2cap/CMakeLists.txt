cmake_minimum_required(VERSION 3.20)
project(L2CAPTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(l2cap_tests L2CAPTests.cpp ../../src/NimBLEL2CAPChannel.cpp)
target_compile_features(l2cap_tests PRIVATE cxx_std_17)
target_include_directories(l2cap_tests PRIVATE stubs ../../src)
target_compile_options(l2cap_tests PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/stubs/Host.h)
target_link_libraries(l2cap_tests PRIVATE Threads::Threads)
add_test(NAME l2cap_errors COMMAND l2cap_tests errors)
add_test(NAME l2cap_early_completion COMMAND l2cap_tests early)
add_test(NAME l2cap_disconnect COMMAND l2cap_tests disconnect)
set_tests_properties(l2cap_errors l2cap_early_completion l2cap_disconnect PROPERTIES TIMEOUT 10)

foreach(case serialize timeout queued_timeout callbacks reconnect ownership lifecycle partial append success pool)
  add_test(NAME l2cap_${case} COMMAND l2cap_tests ${case})
  set_tests_properties(l2cap_${case} PROPERTIES TIMEOUT 10)
endforeach()

add_executable(l2cap_inline_tests L2CAPTests.cpp ../../src/NimBLEL2CAPChannel.cpp)
target_compile_features(l2cap_inline_tests PRIVATE cxx_std_17)
target_compile_definitions(l2cap_inline_tests PRIVATE CONFIG_NIMBLE_CPP_L2CAP_DEFERRED_READ_CALLBACKS=0)
target_include_directories(l2cap_inline_tests PRIVATE stubs ../../src)
target_compile_options(l2cap_inline_tests PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/stubs/Host.h)
target_link_libraries(l2cap_inline_tests PRIVATE Threads::Threads)
add_test(NAME l2cap_inline_errors COMMAND l2cap_inline_tests errors)

add_test(NAME l2cap_inline_lifecycle COMMAND l2cap_inline_tests lifecycle)
