cmake_minimum_required(VERSION 3.20)
project(L2CAPTests LANGUAGES CXX)
enable_testing()
add_executable(l2cap_send_tests SendErrorsTests.cpp ../../src/NimBLEL2CAPChannel.cpp)
target_compile_features(l2cap_send_tests PRIVATE cxx_std_17)
target_include_directories(l2cap_send_tests PRIVATE stubs ../../src)
target_compile_options(l2cap_send_tests PRIVATE -UNDEBUG -include ${CMAKE_CURRENT_SOURCE_DIR}/stubs/Host.h)
foreach(case errors partial ownership append success stalled)
  add_test(NAME l2cap_send_${case} COMMAND l2cap_send_tests ${case})
endforeach()
