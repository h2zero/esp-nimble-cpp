112 public:
113 class Lock {
114 static ble_npl_mutex m_mutex;
115 static std::atomic<bool> m_initialized;
116 static std::atomic<int> m_activeCount;
117 bool m_locked{false};
118
119 public:
120 Lock() {
121 // Increment active count to signal that a constructor is running
122 m_activeCount.fetch_add(1);
123
124 if (m_initialized.load()) {
125 m_locked = (ble_npl_mutex_pend(&m_mutex, BLE_NPL_TIME_FOREVER) == BLE_NPL_OK);
126
127 // Post-lock check: If deinit flipped the flag while we were waiting in line,
128 // we must abort, give it back, and act un-locked.
129 if (!m_initialized.load() && m_locked) {
130 ble_npl_mutex_release(&m_mutex);
131 m_locked = false;
132 }
133 }
134
135 // Decrement active count to signal that the constructor has finished processing
136 m_activeCount.fetch_sub(1);
137 }
138
139 ~Lock() {
140 if (m_locked) {
141 ble_npl_mutex_release(&m_mutex);
142 }
143 }
144
145 static bool init() {
146 if (!m_initialized.load()) {
147 if (ble_npl_mutex_init(&m_mutex) == BLE_NPL_OK) {
148 m_initialized.store(true);
149 }
150 }
151 return m_initialized.load();
152 }
153
154 static void deinit() {
155 if (m_initialized.load()) {
156 // Prevent any new instances from acquiring the lock while we deinitialize
157 m_initialized.store(false);
158
159 // Wait for any active users to finish
160 while (m_activeCount.load() > 0) {
161 ble_npl_time_delay(1);
162 }
163
164 // Acquire the mutex to ensure no other threads are holding it before deinitializing.
165 if (ble_npl_mutex_pend(&m_mutex, BLE_NPL_TIME_FOREVER) == BLE_NPL_OK) {
166 // Deinitialize the mutex now that we have exclusive ownership and no active users.
167 ble_npl_mutex_deinit(&m_mutex);
168 }
169 }
170 }
171 };

453 T getValue(time_t* timestamp = nullptr, bool skipSizeCheck = false) const {
454 Lock lock;
455 if (timestamp != nullptr) {
456# if MYNEWT_VAL(NIMBLE_CPP_ATT_VALUE_TIMESTAMP_ENABLED)
457 *timestamp = m_timestamp;
458# else
459 *timestamp = 0;
460# endif
461 }
462
463 if (!skipSizeCheck && size() < sizeof(T)) {
464 return T();
465 }
466 return *(reinterpret_cast<const T*>(m_attr_value));
467 }

490 NimBLEAttValue& operator+=(const NimBLEAttValue& source) {
491 Lock lock; // Recursive; keeps the source data and size consistent while appending
492 if (&source == this) {
493 // append() may reallocate the buffer, so append from a copy of our own data
494 std::vector<uint8_t> copy(m_attr_value, m_attr_value + m_attr_len);
495 return append(copy.data(), copy.size());
496 }
497 return append(source.data(), source.size());
498 }
